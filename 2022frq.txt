public int countIncreasingCols() {
    int count = 0;
    boolean isIncreasing = true;
    for (int i = 0; i < grid.length; i++) {
        for (int j = 0; j < grid[0].length - 1; j++) {
            if (grid[j][i] > grid[j+1][i]) {
                isIncreasing = false;
            }
        }
        if(isIncreasing) {
            count++;
        }
    }
    return count;
}
